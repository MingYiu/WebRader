    -- ========== Scan Weapon Entities ==========
    -- Returns list of {x,y,z,class,short_name} for every weapon entity on the map.
    -- Caches per-frame to avoid re-iteration cost.
    local weapon_scan_cache = { ts = 0, list = {} }
    local function scan_weapon_entities(force_refresh)
        local now = globals.RealTime() or 0
        if not force_refresh and weapon_scan_cache.ts and (now - weapon_scan_cache.ts) < 0.25 then
            return weapon_scan_cache.list
        end
        local list = {}
        for _, cls in ipairs(WEAPON_CLASSES) do
            local found = entities.FindByClass(cls)
            if found then
                for _, e in ipairs(found) do
                    if e then
                        local ok, pos = pcall(function() return e:GetAbsOrigin() end)
                        if ok and pos then
                            local x, y, z
                            if type(pos) == "userdata" then
                                x, y, z = pos.x, pos.y, pos.z
                            elseif pos[1] then
                                x, y, z = pos[1], pos[2], pos[3]
                            end
                            if x and y and z then
                                table.insert(list, {
                                    x = x, y = y, z = z,
                                    class = cls,
                                    short = WEAPON_NAME_MAP[cls] or cls,
                                    is_c4 = (cls == "C_C4"),
                                })
                            end
                        end
                    end
                end
            end
        end
        weapon_scan_cache.ts = now
        weapon_scan_cache.list = list
        return list
    end

    -- ========== UDP Client ==========
    local udp_socket = nil

    local function init_udp()
        if udp_socket then return true end
        
        local ok, sock = pcall(function()
            return network.Socket("UDP")
        end)
        
        if ok and sock then
            udp_socket = sock
            print("[Radar] UDP socket created!")
            return true
        else
            print("[Radar] UDP socket failed: " .. tostring(sock))
            return false
        end
    end

    -- ========== JSON Encoder ==========
    local function escape(s)
        if type(s) ~= "string" then return "" end
        s = string.gsub(s, '\\', '\\\\')
        s = string.gsub(s, '"', '\\"')
        s = string.gsub(s, '\n', '\\n')
        s = string.gsub(s, '\r', '\\r')
        s = string.gsub(s, '\t', '\\t')
        return s
    end

    local function json_encode(v)   
        if v == nil then return "null"
        elseif v == true then return "true"
        elseif v == false then return "false"
        elseif type(v) == "number" then return string.format("%.2f", v)
        elseif type(v) == "string" then return '"' .. escape(v) .. '"'
        elseif type(v) == "table" then
            local is_arr = (#v > 0)
            if is_arr then
                local t = {}
                for i = 1, #v do table.insert(t, json_encode(v[i])) end
                return "[" .. table.concat(t, ",") .. "]"
            else
                local t = {}
                for k, val in pairs(v) do
                    if type(k) == "string" then
                        table.insert(t, '"' .. escape(k) .. '":' .. json_encode(val))
                    end
                end
                return "{" .. table.concat(t, ",") .. "}"
            end
        end
        return "null"
    end

    -- ========== Send Data via UDP ==========
    local last_send_log = 0
    local send_count = 0
    local function send_data(data)
        if not udp_socket then
            if not init_udp() then return false end
        end

        local json_str = json_encode(data)
        local ok, size = pcall(function()
            return udp_socket:SendTo(CONFIG.server_ip, CONFIG.server_port, json_str)
        end)

        -- ?????????????? socket????????????????????????
        if not ok then
            print("[Radar] Socket send failed, resetting: " .. tostring(size))
            udp_socket = nil
            return false
        end

        -- Log every 30 sends (~1 sec at 30fps)
        send_count = send_count + 1
        if send_count - last_send_log >= 30 then
            last_send_log = send_count
            local player_count = data.players and #data.players or 0
            print(string.format("[Radar] Sent #%d: %d players, %d bytes", send_count, player_count, #json_str))
        end

        return ok and size and size > 0
    end

    -- ========== Get Team Scores ==========
    local function get_team_scores()
        local t_score = 0
        local ct_score = 0
        local my_team = 2

        -- Get local player team
        local me = entities.GetLocalPlayer()
        if me then
            pcall(function()
                my_team = me:GetPropInt("m_iTeamNum") or 2
            end)
        end

        -- CS2 method 1: try CCSGameRulesProxy (has all team scores on the entity)
        pcall(function()
            local gr_entities = entities.FindByClass("CCSGameRulesProxy")
            if gr_entities and gr_entities[1] then
                local gr = gr_entities[1]
                -- Try a few CS2 score prop names
                pcall(function()
                    local s = gr:GetPropInt("m_iMatchScore_CT") or gr:GetPropInt("m_ctScore") or gr:GetPropInt("m_scoreCT")
                    if s and s >= 0 then ct_score = s end
                end)
                pcall(function()
                    local s = gr:GetPropInt("m_iMatchScore_T") or gr:GetPropInt("m_tScore") or gr:GetPropInt("m_scoreT")
                    if s and s >= 0 then t_score = s end
                end)
            end
        end)

        -- CS2 method 2: try CCSTeam entities (older CS:GO style ??may still work in CS2)
        if t_score == 0 and ct_score == 0 then
            pcall(function()
                local teams = entities.FindByClass("CCSTeam")
                if teams then
                    for _, team in ipairs(teams) do
                        if team then
                            pcall(function()
                                local team_num = team:GetPropInt("m_iTeamNum")
                                local score = team:GetPropInt("m_iScore")
                                              or team:GetPropInt("m_score")
                                              or team:GetPropInt("m_iMatchScore")
                                              or 0
                                if team_num == 2 then
                                    t_score = score
                                elseif team_num == 3 then
                                    ct_score = score
                                end
                            end)
                        end
                    end
                end
            end)
        end

        -- Return based on local player's team
        -- team_score = my team's score, enemy_score = enemy team's score
        if my_team == 2 then
            return t_score, ct_score, ct_score, t_score  -- I'm T: (my=T, enemy=CT, ct, t)
        else
            return ct_score, t_score, ct_score, t_score  -- I'm CT
        end
    end

    -- ========== Get Round Info ==========
    local last_cached_round = 0
    local last_cached_phase = "live"
    local cached_round_time = 115
    local cached_bomb_time = 40
    local cached_bomb_planted = false

    local function get_round_info()
        local round_time = cached_round_time
        local bomb_time = cached_bomb_time
        local bomb_planted = cached_bomb_planted
        local phase = "live"
        local round_number = 0

        -- Try to get from game rules
        pcall(function()
            local game_rules = entities.FindByClass("CCSGameRulesProxy")
            if game_rules and game_rules[1] then
                local gr = game_rules[1]
                local warmup = gr:GetPropInt("m_bIsValveDS") or 0
                if warmup == 1 then
                    phase = "warmup"
                end

                -- Check if bomb is planted
                bomb_planted = gr:GetPropInt("m_bBombPlanted") == 1

                -- CS2 round number lives on CCSGameRules under various names
                pcall(function()
                    local rn = gr:GetPropInt("m_iRoundNum")  -- CS2 (modern)
                    if rn and rn > 0 then round_number = rn end
                end)
                pcall(function()
                    if round_number == 1 then
                        local rn = gr:GetPropInt("m_totalRoundsPlayed")
                        if rn and rn >= 0 then round_number = rn + 1 end
                    end
                end)
                pcall(function()
                    if round_number == 1 then
                        local rn = gr:GetPropInt("m_nRound")
                        if rn and rn > 0 then round_number = rn end
                    end
                end)
            end
        end)

        -- Get round number from CDirector (legacy / Source)
        pcall(function()
            if round_number == 1 then
                local cd = entities.FindByClass("CDirector")
                if cd and cd[1] then
                    local rn = cd[1]:GetPropInt("m_iRoundNumber")
                    if rn and rn > 0 then
                        round_number = rn
                    end
                end
            end
        end)

        -- Increment round change tracking (used to reset round timer)
        local phase_changed = phase ~= last_cached_phase
        if round_number ~= last_cached_round or phase_changed then
            last_cached_round = round_number
            last_cached_phase = phase
            cached_round_time = 115  -- Reset timer on new round
            cached_bomb_time = 40
            cached_bomb_planted = false
        end

        -- Update timer (decrement by ~0.5s each frame)
        if not bomb_planted then
            cached_round_time = math.max(0, cached_round_time - 0.5)
            round_time = cached_round_time
        else
            cached_bomb_time = math.max(0, cached_bomb_time - 0.5)
            bomb_time = cached_bomb_time
        end

        return round_time, bomb_time, bomb_planted, phase, round_number
    end

    -- ========== Get Player Data ==========
    local function dist3(ax, ay, az, bx, by, bz)
        local dx, dy, dz = ax - bx, ay - by, az - bz
        return math.sqrt(dx * dx + dy * dy + dz * dz)
    end

    -- Returns the nearest weapon within WEAPON_HOLD_RADIUS, or nil.
    -- Designed for held weapons: a player's held gun sits ~0-1 units from pawn origin.
    -- Ground weapons fall outside this radius (they're at floor level).
    local WEAPON_HOLD_RADIUS = 80  -- generous; player bounds in CS2 ~ 72 units tall
    local function nearest_weapon_for(x, y, z, weapons)
        local best, best_dist = nil, WEAPON_HOLD_RADIUS
        for _, w in ipairs(weapons) do
            local d = dist3(x, y, z, w.x, w.y, w.z)
            if d < best_dist then
                best_dist = d
                best = w
            end
        end
        return best
    end

    local function get_player_data(ent, is_local, my_team, weapon_list)
        local pos = {x = 0, y = 0, z = 0}
        local team = 2
        local hp = 100
        local weapon = "Unknown"
        local armor = 0
        local is_alive = 1
        local has_bomb = 0
        local player_name = "Player"
        local steamid = "0"
        local is_bot = 1
        
        pcall(function()
            local o = ent:GetAbsOrigin()
            if o then 
                if type(o) == "userdata" then
                    pos = {x = o.x or 0, y = o.y or 0, z = o.z or 0}
                elseif o[1] then
                    pos = {x = o[1] or 0, y = o[2] or 0, z = o[3] or 0}
                end
            end
        end)
        
        -- CS2: Try multiple ways to get team number
        pcall(function()
            if ent.GetTeamNumber then
                team = ent:GetTeamNumber() or 2
            end
        end)
        
        if team == 2 then
            pcall(function()
                team = ent:GetPropInt("m_iTeamNum") or 2
            end)
        end
        
        if team == 2 then
            pcall(function()
                team = ent:GetPropInt("m_nTeamNum") or 2
            end)
        end
        
        pcall(function()
            hp = ent:GetPropInt("m_iHealth") or 100
        end)
        
        pcall(function()
            armor = ent:GetPropInt("m_ArmorValue") or 0
        end)

        local has_helmet = 0
        pcall(function()
            -- CS2 helmet flag (multiple known paths)
            local h = ent:GetPropInt("m_bHasHelmet")
            if h == 1 then has_helmet = 1 end
        end)
        pcall(function()
            if has_helmet == 0 then
                local h = ent:GetPropInt("m_bHelmetEnabled")
                if h == 1 then has_helmet = 1 end
            end
        end)
        pcall(function()
            if has_helmet == 0 then
                local h = ent:GetPropInt("m_iItemDefinitionIndex")  -- placeholder; some weapon properties
                -- Skip if prop doesn't exist; just ignore
            end
        end)
        
        pcall(function()
            is_alive = (ent:IsAlive() == true) and 1 or 0
        end)
        
        pcall(function()
            has_bomb = (ent:GetPropInt("m_bHasDefuser") == 1) and 1 or 0
        end)
        
        -- Try to get player name (for CS2, might need controller)
        pcall(function()
            player_name = ent:GetName() or "Player"
        end)

        -- Try to get SteamID via multiple known methods (Aimware/Source Engine + CS2 netvars)
        -- Bot detection: bot's steamid = 0; real player has 64-bit SteamID
        pcall(function()
            -- Method 1: Direct engine API (Aimware/Source)
            if ent.GetSteamID then
                local sid = ent:GetSteamID()
                if sid and type(sid) == "string" and sid ~= "0" and sid ~= "BOT" then
                    steamid = sid
                    is_bot = 0
                end
            end
        end)
        pcall(function()
            -- Method 2: CS2 netvar m_steamID (string, e.g. "STEAM_0:0:12345")
            if steamid == "0" then
                local sid = ent:GetPropString("m_steamID")
                if sid and sid ~= "" and sid ~= "0" and sid ~= "BOT" then
                    steamid = sid
                    is_bot = 0
                end
            end
        end)
        pcall(function()
            -- Method 3: CS2 netvar m_iAccountID (uint32, lower 32 bits of SteamID64)
            if steamid == "0" then
                local aid = ent:GetPropInt("m_iAccountID")
                if aid and aid > 0 and aid ~= 0xFFFFFFFF then
                    -- Convert AccountID to SteamID64: base = 76561197960265728
                    steamid = tostring(0x0110000100000000ULL + aid)
                    is_bot = 0
                end
            end
        end)
        pcall(function()
            -- Method 4: From controller (CCSPlayerPawn -> m_hController -> CCSPlayerController)
            if steamid == "0" then
                local controller = ent:GetPropEntity("m_hController")
                if controller then
                    if controller.GetSteamID then
                        local sid = controller:GetSteamID()
                        if sid and type(sid) == "string" and sid ~= "0" and sid ~= "BOT" then
                            steamid = sid
                            is_bot = 0
                        end
                    end
                    if steamid == "0" then
                        local sid = controller:GetPropString("m_steamID")
                        if sid and sid ~= "" and sid ~= "0" and sid ~= "BOT" then
                            steamid = sid
                            is_bot = 0
                        end
                    end
                    if steamid == "0" then
                        local aid = controller:GetPropInt("m_iAccountID")
                        if aid and aid > 0 and aid ~= 0xFFFFFFFF then
                            steamid = tostring(0x0110000100000000ULL + aid)
                            is_bot = 0
                        end
                    end
                end
            end
        end)
        pcall(function()
            -- Method 5: m_bIsBot / m_fFlags / is_bot check (CS2)
            if is_bot == 1 and steamid == "0" then
                local bot_flag = ent:GetPropInt("m_bIsBot")
                if bot_flag == 1 then
                    is_bot = 1
                elseif bot_flag == 0 then
                    -- m_bIsBot = 0 doesn't necessarily mean real, but worth keeping
                end
            end
        end)
        
pcall(function()
            -- Weapon association by position proximity.
            -- Since aimware Lua here cannot read m_hActiveWeapon (all prop APIs
            -- return NO_METHOD on this build), we use the only remaining signal:
            -- the held weapon entity's world position. A held weapon renders at
            -- approximately the same coordinates as the owning pawn origin.
            -- Limitation: dead players' dropped weapons will be at their corpse,
            -- so we only assign when alive. We also prefer C4 over regular weapons.
            if is_alive == 1 and weapon_list and weapon_list.nearest_weapon then
                local best = weapon_list.nearest_weapon(ent)
                if best then
                    weapon = best.short
                    if best.is_c4 then
                        has_bomb = 1
                    end
                end
            end
            -- Fallback: keep previous frame's weapon across brief hiccups
            if weapon == "Unknown" then
                local entIndex = ent:GetIndex()
                local prev = weapon_frame_cache[entIndex]
                if prev and prev ~= "Unknown" then weapon = prev end
            elseif weapon ~= "Unknown" then
                weapon_frame_cache[ent:GetIndex()] = weapon
            end
        end)
        
        return {
            index = ent:GetIndex(),
            name = player_name,
            team = team,
            is_local = is_local and 1 or 0,
            is_team = (team == my_team) and 1 or 0,
            is_alive = is_alive,
            is_bot = is_bot,
            steamid = steamid,
            x = pos.x,
            y = pos.y,
            z = pos.z,
            view_x = 0,
            view_y = 0,
            hp = hp,
            armor = armor,
            has_helmet = has_helmet,
            weapon = weapon,
            has_bomb = has_bomb
        }
    end

    -- ========== Entity Class Discovery ==========
    -- Call this manually in console: radar_discover_entities()
    function radar_discover_entities()
        print("[Radar] Discovering entity classes...")
        
        -- Common player class names to try
        local class_names = {
            "CCSPlayer", "C_CSPlayer", "CSPlayer", "Player",
            "CTFPlayer", "TFPlayer", "TeamFortressPlayer",
            "CBasePlayer", "BasePlayer", "CPlayer", "PlayerEnt"
        }
        
        local found_players = {}
        
        for _, class_name in ipairs(class_names) do
            local entities_list = entities.FindByClass(class_name)
            if entities_list and #entities_list > 0 then
                print("[Radar] Found " .. #entities_list .. " entities with class: " .. class_name)
                for i, ent in ipairs(entities_list) do
                    if ent then
                        local idx = ent:GetIndex()
                        print("  Entity[" .. i .. "]: Index=" .. idx)
                        pcall(function()
                            local alive = ent:IsAlive()
                            print("    IsAlive=" .. tostring(alive))
                        end)
                    end
                end
                found_players[class_name] = entities_list
            end
        end
        
        -- Try wildcard search
        print("[Radar] Trying wildcard search...")
        for _, class_name in ipairs(class_names) do
            local wildcard = class_name .. "*"
            local entities_list = entities.FindByClass(wildcard)
            if entities_list and #entities_list > 0 then
                print("[Radar] Wildcard " .. wildcard .. " found " .. #entities_list .. " entities")
            end
        end
        
    -- Try entities.GetAll() if available
    local getall = entities.GetAll
    if getall then
        print("[Radar] Trying entities.GetAll()...")
        local all_entities = getall()
        if all_entities then
            print("[Radar] GetAll() returned " .. #all_entities .. " entities")
        end
    else
        print("[Radar] GetAll() not available")
    end
        
        -- Try iterating by index using FindByIndex
        print("[Radar] Trying direct index iteration (0-63)...")
        local player_indices = {}
        
        -- Try different ways to get entity by index
        local get_by_index = entities.FindByIndex or entities.GetByIndex or entities.Get
        if get_by_index then
            for i = 0, 63 do
                local ent = get_by_index(i)
                if ent then
                    local alive = pcall_wrap(function() return ent:IsAlive() end)
                    if alive then
                        local name = pcall_wrap(function() return ent:GetName() end)
                        local class = pcall_wrap(function() return ent:GetClass() end)
                        print("  Index[" .. i .. "]: Name=" .. tostring(name) .. ", Class=" .. tostring(class) .. ", Alive=" .. tostring(alive))
                        table.insert(player_indices, {index = i, name = name, class = class, alive = alive})
                    end
                end
            end
            print("[Radar] Found " .. #player_indices .. " alive entities by index")
        else
            print("[Radar] FindByIndex/GetByIndex not available")
        end
        
        print("[Radar] Discovery complete!")
        return found_players, player_indices
    end

    -- Helper function for safe pcall
    function pcall_wrap(fn)
        local ok, result = pcall(fn)
        if ok then return result else return nil end
    end
